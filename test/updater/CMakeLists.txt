cmake_minimum_required(VERSION 3.16)
project(UpdateProgressTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(update_progress_test ../UpdateProgressTest.cpp)
target_include_directories(update_progress_test PRIVATE ../..)
target_compile_options(update_progress_test PRIVATE /utf-8)
target_link_libraries(update_progress_test PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME update_progress COMMAND update_progress_test)
set_tests_properties(update_progress PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
