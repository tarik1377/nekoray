cmake_minimum_required(VERSION 3.22)
project(TunBypassTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
get_filename_component(GR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_executable(tun_bypass_test "${GR_SOURCE_DIR}/test/TunBypassTest.cpp" "${GR_SOURCE_DIR}/ui/TunBypassDialog.cpp")
target_include_directories(tun_bypass_test PRIVATE "${GR_SOURCE_DIR}")
target_link_libraries(tun_bypass_test PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
    target_compile_options(tun_bypass_test PRIVATE /utf-8)
endif()
add_test(NAME tun_bypass_test COMMAND tun_bypass_test)
set_tests_properties(tun_bypass_test PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
