cmake_minimum_required(VERSION 3.16)
project(greenrhythm_settings_widgets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
foreach(target settings_preview settings_page_test maintenance_panel_test)
    if(target STREQUAL settings_preview)
        set(entry SettingsPreview.cpp)
    elseif(target STREQUAL settings_page_test)
        set(entry SettingsPageTest.cpp)
    else()
        set(entry MaintenancePanelTest.cpp)
    endif()
    add_executable(${target}
        ${APP_ROOT}/test/${entry} ${APP_ROOT}/main/AppFont.cpp
        ${APP_ROOT}/ui/MainShell.cpp ${APP_ROOT}/ui/MainShell.hpp
        ${APP_ROOT}/ui/Icons.cpp ${APP_ROOT}/ui/ServerCardDelegate.cpp
        ${APP_ROOT}/res/neko.qrc)
    target_include_directories(${target} PRIVATE ${APP_ROOT})
    target_link_libraries(${target} PRIVATE Qt6::Widgets Qt6::Svg)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
    if(NOT target STREQUAL settings_preview)
        add_test(NAME ${target} COMMAND ${target})
        set_tests_properties(${target} PROPERTIES WORKING_DIRECTORY ${APP_ROOT} ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endforeach()
